  <script type="text/babel">
    const { useState, useEffect } = React;

    // --- 아이콘 컴포넌트 ---
    const Home = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="m3 9 9-7 9 7v11a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2z"/><polyline points="9 22 9 12 15 12 15 22"/></svg>;
    const ArrowRight = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M5 12h14"/><path d="m12 5 7 7-7 7"/></svg>;
    const CheckCircle = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg>;
    const RotateCcw = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M3 12a9 9 0 1 0 9-9 9.75 9.75 0 0 0-6.74 2.74L3 8"/><path d="M3 3v5h5"/></svg>;
    const X = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M18 6 6 18"/><path d="m6 6 12 12"/></svg>;
    const XCircle = ({ className }) => <svg className={className} xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><circle cx="12" cy="12" r="10"/><path d="m15 9-6 6"/><path d="m9 9 6 6"/></svg>;

    function App() {
      // --- 상태 관리 ---
      const [screen, setScreen] = useState('home'); // 'home', 'game'
      const [mode, setMode] = useState('basic'); // 'basic', 'advanced', 'custom'
      
      const [num1, setNum1] = useState(0);
      const [num2, setNum2] = useState(0);
      
      const [step, setStep] = useState(1); // 1: 분해하기, 2: 빼기, 3: 세로셈
      const [blocks, setBlocks] = useState([]);
      
      const [hasAnsweredOx, setHasAnsweredOx] = useState(false); // O/X 팝업 상태
      const [oxFeedback, setOxFeedback] = useState(''); // O/X 피드백

      const [inputs, setInputs] = useState({
        regroupTen: '',
        regroupOne: '',
        ansTen: '',
        ansOne: ''
      });
      
      const [errorFields, setErrorFields] = useState([]);
      const [showErrorModal, setShowErrorModal] = useState(false);
      const [showSuccessModal, setShowSuccessModal] = useState(false);

      // --- 직접 식 만들기(Custom) 관련 상태 ---
      const [showCustomModal, setShowCustomModal] = useState(false);
      const [customNum1, setCustomNum1] = useState('');
      const [customNum2, setCustomNum2] = useState('');
      const [customError, setCustomError] = useState('');
      
      // --- 이전 단계로 돌아가기 ---
      const handleGoBack = () => {
        if (step === 3) {
          setStep(2);
        } else if (step === 2) {
          setStep(1);
          // 모형 초기화 (1단계로 복구)
          const initialBlocks = [];
          let idCounter = 1;
          const tensCount = Math.floor(num1 / 10);
          const onesCount = num1 % 10;
          for (let i = 0; i < tensCount; i++) {
            initialBlocks.push({ id: idCounter++, type: 'ten', status: 'active' });
          }
          for (let i = 0; i < onesCount; i++) {
            initialBlocks.push({ id: idCounter++, type: 'one', status: 'active' });
          }
          setBlocks(initialBlocks);
        }
      };

      // --- 문제 생성 및 초기화 ---
      const generateProblem = (currentMode) => {
        let n1, n2;
        if (currentMode === 'basic') {
          // 두 자리 수 - 한 자리 수 (받아내림 필수)
          do {
            n1 = Math.floor(Math.random() * 80) + 11; // 11 ~ 90
            n2 = Math.floor(Math.random() * 8) + 2;   // 2 ~ 9
          } while (n1 % 10 >= n2 % 10);
        } else {
          // 두 자리 수 - 두 자리 수 (받아내림 필수)
          do {
            n1 = Math.floor(Math.random() * 70) + 21; // 21 ~ 90
            n2 = Math.floor(Math.random() * 60) + 11; // 11 ~ 70
          } while (n1 % 10 >= n2 % 10 || n1 <= n2);
        }
        
        launchGameWithNumbers(n1, n2, currentMode);
      };

      // 공통 게임 런처
      const launchGameWithNumbers = (n1, n2, gameMode) => {
        setNum1(n1);
        setNum2(n2);
        setMode(gameMode);
        setStep(1);
        setHasAnsweredOx(false);
        setOxFeedback('');
        setInputs({ regroupTen: '', regroupOne: '', ansTen: '', ansOne: '' });
        setErrorFields([]);
        
        // 모형 초기화
        const initialBlocks = [];
        let idCounter = 1;
        
        const tensCount = Math.floor(n1 / 10);
        const onesCount = n1 % 10;
        
        for (let i = 0; i < tensCount; i++) {
          initialBlocks.push({ id: idCounter++, type: 'ten', status: 'active' });
        }
        for (let i = 0; i < onesCount; i++) {
          initialBlocks.push({ id: idCounter++, type: 'one', status: 'active' });
        }
        setBlocks(initialBlocks);
      };

      const startGame = (selectedMode) => {
        setMode(selectedMode);
        generateProblem(selectedMode);
        setScreen('game');
      };

      // --- 직접 만든 식 유효성 검사 및 시작 ---
      const handleCustomSubmit = (e) => {
        e.preventDefault();
        const n1 = parseInt(customNum1);
        const n2 = parseInt(customNum2);

        if (isNaN(n1) || isNaN(n2)) {
          setCustomError('숫자를 올바르게 입력해주세요!');
          return;
        }
        if (n1 < 11 || n1 > 99) {
          setCustomError('첫 번째 수는 11부터 99까지의 두 자리 수여야 해요!');
          return;
        }
        if (n2 < 1 || n2 > 99) {
          setCustomError('두 번째 수는 1부터 99까지의 수여야 해요!');
          return;
        }
        if (n1 <= n2) {
          setCustomError('빼어지는 수가 빼는 수보다 항상 더 커야 해요!');
          return;
        }
        // 받아내림 핵심 검증 (num1의 일의 자리가 num2의 일의 자리보다 작아야 함)
        if (n1 % 10 >= n2 % 10) {
          setCustomError(`일의 자리에서 받아내림이 없어요! 일의 자리(${n1 % 10})가 빼는 일의 자리(${n2 % 10})보다 작아야 해요.`);
          return;
        }

        // 성공 검증 시 게임 화면 실행
        setCustomError('');
        setShowCustomModal(false);
        setScreen('game');
        
        // 난이도 모드 결정
        const determinedMode = n2 < 10 ? 'basic' : 'advanced';
        launchGameWithNumbers(n1, n2, determinedMode);
      };

      // --- 단계별 액션 ---
      // 1단계: 십모형 쪼개기
      const handleDecompose = (id) => {
        if (step !== 1) return;
        
        // [안전 잠금] 이미 어떤 십모형이 쪼개지고 있는 중이라면 추가 입력을 무시합니다.
        if (blocks.some(b => b.status === 'decomposing')) return;

        const block = blocks.find(b => b.id === id);
        if (block.type !== 'ten') return;

        // 1. 애니메이션 시작 (노란 불빛으로 빛나며 wiggle)
        setBlocks(prev => prev.map(b => b.id === id ? { ...b, status: 'decomposing' } : b));
        
        // 2. 1.5초 후 십모형 제거 및 일모형 10개 생성 후 단계 이동
        setTimeout(() => {
          setBlocks(prev => {
            const withoutDecomposed = prev.filter(b => b.id !== id);
            const newOnes = [];
            let newId = Math.max(...prev.map(b => b.id)) + 1;
            for (let i = 0; i < 10; i++) {
              newOnes.push({ id: newId++, type: 'one', status: 'new' });
            }
            return [...withoutDecomposed, ...newOnes];
          });
          setStep(2); // 쪼개기 성공 시 바로 2단계로 강제 전환하여 중복 쪼개기 원천 차단
        }, 1500);
      };

      // 2단계: 모형 빼기 (선택)
      const handleToggleRemove = (id) => {
        if (step !== 2) return;
        setBlocks(prev => prev.map(b => {
          if (b.id === id) {
            return { ...b, status: b.status === 'removed' ? 'active' : 'removed' };
          }
          return b;
        }));
      };

      // 2단계 정답 확인
      const checkStep2 = () => {
        const removedTens = blocks.filter(b => b.type === 'ten' && b.status === 'removed').length;
        const removedOnes = blocks.filter(b => b.type === 'one' && b.status === 'removed').length;
        
        const targetTens = Math.floor(num2 / 10);
        const targetOnes = num2 % 10;
        
        if (removedTens === targetTens && removedOnes === targetOnes) {
          setStep(3);
        } else {
          triggerError([]);
        }
      };

      // 3단계: 세로셈 정답 확인
      const checkStep3 = () => {
        const expectedRegroupTen = Math.floor(num1 / 10) - 1;
        const expectedRegroupOne = 10;
        const expectedAns = num1 - num2;
        const expectedAnsTen = Math.floor(expectedAns / 10) === 0 ? '' : String(Math.floor(expectedAns / 10));
        const expectedAnsOne = String(expectedAns % 10);

        const errors = [];
        if (inputs.regroupTen !== String(expectedRegroupTen)) errors.push('regroupTen');
        if (inputs.regroupOne !== String(expectedRegroupOne)) errors.push('regroupOne');
        if (inputs.ansTen !== expectedAnsTen && !(expectedAnsTen === '' && inputs.ansTen === '0')) errors.push('ansTen');
        if (inputs.ansOne !== expectedAnsOne) errors.push('ansOne');

        if (errors.length > 0) {
          triggerError(errors);
        } else {
          setShowSuccessModal(true);
        }
      };

      const triggerError = (fields) => {
        setErrorFields(fields);
        setShowErrorModal(true);
      };

      const closeErrorModal = () => {
        setShowErrorModal(false);
        // 모달 닫힌 후 2초간 번쩍임 지속
        setTimeout(() => {
          setErrorFields([]);
        }, 2000);
      };

      // --- 화면 렌더링 ---
      if (screen === 'home') {
        return (
          <div className="min-h-screen bg-sky-50 flex flex-col items-center justify-center p-6 font-sans relative pb-28">
            <div className="bg-white p-10 rounded-3xl shadow-xl max-w-lg w-full text-center border-4 border-indigo-100">
              <div className="text-5xl mb-4">🚀</div>
              <h1 className="text-3xl font-bold text-gray-800 mb-2 animate-bounce">초등 2학년 수학</h1>
              <h2 className="text-2xl font-bold text-indigo-600 mb-8">받아내림이 있는 뺄셈</h2>
              
              <div className="space-y-4 flex flex-col">
                <button 
                  onClick={() => startGame('basic')}
                  className="w-full py-4 bg-green-100 hover:bg-green-200 text-green-800 rounded-2xl text-xl font-bold transition-colors border-2 border-green-300 shadow-sm"
                >
                  🌱 기본편 (두 자리 수 - 한 자리 수)
                </button>
                <button 
                  onClick={() => startGame('advanced')}
                  className="w-full py-4 bg-indigo-100 hover:bg-indigo-200 text-indigo-800 rounded-2xl text-xl font-bold transition-colors border-2 border-indigo-300 shadow-sm"
                >
                  🚀 심화편 (두 자리 수 - 두 자리 수)
                </button>
                <button 
                  onClick={() => {
                    setCustomNum1('');
                    setCustomNum2('');
                    setCustomError('');
                    setShowCustomModal(true);
                  }}
                  className="w-full py-4 bg-amber-100 hover:bg-amber-200 text-amber-800 rounded-2xl text-xl font-bold transition-colors border-2 border-amber-300 shadow-sm flex items-center justify-center gap-2"
                >
                  ✏️ 내가 직접 식 만들기
                </button>
              </div>
            </div>

            {/* 직접 식 만들기 설정 팝업(모달) */}
            {showCustomModal && (
              <div className="fixed inset-0 bg-black/60 flex items-center justify-center z-50 backdrop-blur-sm">
                <div className="bg-white p-8 rounded-3xl shadow-2xl text-center max-w-md w-full mx-4 border-4 border-amber-400 animate-pop-in">
                  <div className="flex justify-between items-center mb-4">
                    <h3 className="text-2xl font-bold text-slate-800">✏️ 직접 식 만들기</h3>
                    <button 
                      onClick={() => setShowCustomModal(false)}
                      className="text-slate-400 hover:text-slate-600"
                    >
                      <X className="w-6 h-6" />
                    </button>
                  </div>

                  <p className="text-sm text-slate-500 mb-6 bg-amber-50 p-3 rounded-xl text-left leading-relaxed">
                    💡 <b>성공 공식 조건!</b><br/>
                    1. 첫째 수는 <b>11~99</b>, 둘째 수보다 커야 해요.<br/>
                    2. <b>일의 자리</b>에서 반드시 빌려오도록(받아내림), 첫째 수의 일의 자리가 더 작아야 합니다! (예: 52 - 18, 43 - 7)
                  </p>

                  <form onSubmit={handleCustomSubmit} className="space-y-6">
                    <div className="flex items-center justify-center gap-4">
                      <div className="flex flex-col items-center">
                        <label className="text-xs font-bold text-slate-500 mb-1">빼어지는 수</label>
                        <input 
                          type="number"
                          placeholder="예: 52"
                          value={customNum1}
                          onChange={(e) => setCustomNum1(e.target.value)}
                          className="w-24 h-16 text-center text-3xl font-extrabold text-slate-800 border-4 border-slate-200 focus:border-amber-400 rounded-2xl outline-none"
                          min="11"
                          max="99"
                          required
                        />
                      </div>
                      <span className="text-4xl font-bold text-slate-400 pt-5">-</span>
                      <div className="flex flex-col items-center">
                        <label className="text-xs font-bold text-slate-500 mb-1">빼는 수</label>
                        <input 
                          type="number"
                          placeholder="예: 18"
                          value={customNum2}
                          onChange={(e) => setCustomNum2(e.target.value)}
                          className="w-24 h-16 text-center text-3xl font-extrabold text-slate-800 border-4 border-slate-200 focus:border-amber-400 rounded-2xl outline-none"
                          min="1"
                          max="99"
                          required
                        />
                      </div>
                    </div>

                    {customError && (
                      <div className="text-sm text-red-500 font-bold bg-red-50 p-3 rounded-xl border border-red-200 leading-relaxed">
                        ⚠️ {customError}
                      </div>
                    )}

                    <div className="flex gap-3">
                      <button 
                        type="button"
                        onClick={() => setShowCustomModal(false)}
                        className="flex-1 py-3 bg-slate-100 hover:bg-slate-200 text-slate-600 font-bold rounded-xl transition-colors"
                      >
                        취소
                      </button>
                      <button 
                        type="submit"
                        className="flex-1 py-3 bg-amber-500 hover:bg-amber-600 text-white font-bold rounded-xl shadow-md transition-colors"
                      >
                        출발하기 🚀
                      </button>
                    </div>
                  </form>
                </div>
              </div>
            )}

            {/* 제작자 표시 (홈 화면) - 가운데 아래 */}
            <div className="fixed bottom-4 left-1/2 -translate-x-1/2 flex items-center bg-white/95 backdrop-blur px-6 py-3 rounded-full shadow-lg border-2 border-sky-200 z-50 transition-transform hover:scale-105">
              <img 
                src="경서 캐릭터(상반신).jpg" 
                alt="어썸오쌤" 
                className="w-12 h-12 rounded-full mr-3 object-cover border border-slate-200 shadow-sm"
                onError={(e) => { 
                  e.target.onerror = null; 
                  e.target.src = 'https://api.dicebear.com/7.x/notionists/svg?seed=Awesome'; 
                }}
              />
              <span className="font-extrabold text-slate-700 text-lg tracking-wide whitespace-nowrap">제작: 어썸오쌤</span>
            </div>
          </div>
        );
      }

      // 모형 분류
      const tensBlocks = blocks.filter(b => b.type === 'ten');
      const onesBlocks = blocks.filter(b => b.type === 'one');

      return (
        <div className="min-h-screen bg-slate-50 p-4 sm:p-8 font-sans flex flex-col items-center pb-28">
          
          {/* 상단 네비게이션 */}
          <div className="w-full max-w-4xl flex justify-between items-center mb-6">
            <div className="flex gap-3">
              <button 
                onClick={() => setScreen('home')}
                className="flex items-center text-slate-600 hover:text-slate-800 font-bold bg-white px-4 py-2 rounded-xl shadow-sm border border-slate-200 transition-colors"
              >
                <Home className="w-5 h-5 mr-2" /> 홈으로
              </button>
              {step > 1 && (
                <button 
                  onClick={handleGoBack}
                  className="flex items-center text-amber-600 hover:text-amber-800 font-bold bg-amber-50 px-4 py-2 rounded-xl shadow-sm border border-amber-300 transition-colors"
                >
                  <RotateCcw className="w-5 h-5 mr-2" /> 이전 단계로
                </button>
              )}
            </div>
            <div className="text-xl font-bold text-slate-700 bg-white px-6 py-2 rounded-full shadow-sm border border-slate-200">
              현재 단계 : {step} / 3
            </div>
          </div>

          {/* 메인 작업 영역 */}
          <div className="w-full max-w-4xl bg-white rounded-3xl shadow-lg p-6 sm:p-10 mb-8">
            
            {/* 가로셈 제목 */}
            <div className="text-center text-4xl sm:text-5xl font-extrabold text-slate-800 mb-8 tracking-widest">
              {num1} - {num2} = {step === 3 && num1 - num2}
            </div>

            {/* 지시문 */}
            <div className="bg-amber-100 border-l-4 border-amber-500 text-amber-800 p-4 rounded-r-lg mb-8 text-lg font-bold flex items-center">
              <span className="text-2xl mr-3">💡</span>
              {step === 1 && "일모형이 부족해요! 초록색 십모형 1개를 눌러 일모형 10개로 쪼개주세요."}
              {step === 2 && `이제 ${num2}을/를 뺄 수 있어요! 뺄 만큼 모형을 눌러 X 표시를 하세요.`}
              {step === 3 && "받아내림을 표시하고, 세로셈의 빈칸을 채워보세요!"}
            </div>

            <div className="flex flex-col lg:flex-row gap-12">
              
              {/* 수 모형 영역 */}
              <div className="flex-1 bg-slate-100 rounded-2xl p-6 border-2 border-slate-200">
                <h3 className="text-center font-bold text-slate-600 mb-4 bg-white py-2 rounded-lg shadow-sm">수 모형 놓기</h3>
                
                <div className="flex justify-center gap-10">
                  {/* 십모형 칸 */}
                  <div className="flex flex-col items-center">
                    <span className="text-green-600 font-bold mb-2">십모형</span>
                    <div className="flex flex-wrap justify-center gap-3 w-32 min-h-[12rem] items-end content-end">
                      {tensBlocks.map(block => {
                        const isAnyDecomposing = blocks.some(b => b.status === 'decomposing');
                        return (
                          <div 
                            key={block.id}
                            onClick={() => step === 1 ? handleDecompose(block.id) : handleToggleRemove(block.id)}
                            className={`
                              w-8 h-32 rounded-md border border-green-600 flex flex-col overflow-hidden relative transition-all duration-200
                              ${block.status === 'active' || block.status === 'new' ? 'bg-green-400 shadow-md' : ''}
                              ${(block.status === 'active' || block.status === 'new') && !isAnyDecomposing && step === 1 ? 'hover:scale-105 cursor-pointer' : ''}
                              ${block.status === 'decomposing' ? 'bg-yellow-300 animate-wiggle-glow cursor-not-allowed' : ''}
                              ${block.status === 'removed' ? 'bg-slate-300 border-slate-400 opacity-60 scale-95' : ''}
                            `}
                          >
                            {/* 10등분 선 */}
                            {Array(10).fill(0).map((_, i) => (
                              <div key={i} className={`flex-1 border-b ${block.status === 'removed' ? 'border-slate-400' : 'border-green-500'} last:border-b-0`} />
                            ))}
                            {/* X 표시 */}
                            {block.status === 'removed' && (
                              <div className="absolute inset-0 flex items-center justify-center">
                                <X className="text-red-500 w-10 h-10 drop-shadow-md" />
                              </div>
                            )}
                          </div>
                        );
                      })}
                    </div>
                  </div>

                  {/* 일모형 칸 */}
                  <div className="flex flex-col items-center">
                    <span className="text-yellow-600 font-bold mb-2">일모형</span>
                    <div className="flex flex-wrap gap-2 w-48 min-h-[12rem] content-start">
                      {onesBlocks.map(block => (
                        <div 
                          key={block.id}
                          onClick={() => step === 2 && handleToggleRemove(block.id)}
                          className={`
                            w-8 h-8 rounded-md border relative flex items-center justify-center transition-all duration-200
                            ${block.status === 'active' ? 'bg-yellow-400 border-yellow-600 shadow-sm' : ''}
                            ${block.status === 'active' && step === 2 ? 'hover:scale-110 cursor-pointer' : ''}
                            ${block.status === 'new' ? 'bg-yellow-400 border-yellow-600 animate-pop-in shadow-sm' : ''}
                            ${block.status === 'removed' ? 'bg-slate-300 border-slate-400 opacity-60 scale-95 cursor-pointer' : ''}
                          `}
                        >
                          {block.status === 'removed' && <X className="text-red-500 w-8 h-8 drop-shadow-md" />}
                        </div>
                      ))}
                    </div>
                  </div>
                </div>

                {/* 2단계 확인 버튼 */}
                {step === 2 && (
                  <div className="mt-8 flex justify-center">
                    <button 
                      onClick={checkStep2}
                      className="bg-blue-500 hover:bg-blue-600 text-white font-bold py-3 px-8 rounded-xl shadow-md flex items-center text-lg transition-transform hover:scale-105 active:scale-95"
                    >
                      <CheckCircle className="mr-2 w-6 h-6" /> 뺄셈 확인하기
                    </button>
                  </div>
                )}
              </div>

              {/* 세로셈 영역 (3단계에서 활성화) */}
              <div className={`flex-1 bg-white rounded-2xl p-6 border-2 flex flex-col items-center justify-center transition-opacity duration-500 ${step === 3 ? 'border-blue-300 opacity-100' : 'border-slate-200 opacity-30 pointer-events-none'}`}>
                 <h3 className="text-center font-bold text-slate-600 mb-6 bg-slate-50 py-2 px-6 rounded-lg shadow-sm">세로셈 완성하기</h3>
                 
                 <div className="text-5xl font-bold text-slate-800 grid grid-cols-3 gap-y-4 gap-x-2 items-center justify-items-center bg-white p-8 rounded-xl shadow-inner border border-slate-100">
                    {/* 1행: 받아내림 적는 곳 */}
                    <div className="col-start-2">
                      <input 
                        type="number" 
                        value={inputs.regroupTen}
                        onChange={(e) => setInputs({...inputs, regroupTen: e.target.value.slice(0, 1)})}
                        className={`w-14 h-14 text-center text-2xl text-red-500 border-2 rounded-lg bg-red-50 focus:outline-none focus:border-red-400 ${errorFields.includes('regroupTen') ? 'flash-error' : 'border-red-200'}`}
                      />
                    </div>
                    <div className="col-start-3">
                      <input 
                        type="number" 
                        value={inputs.regroupOne}
                        onChange={(e) => setInputs({...inputs, regroupOne: e.target.value.slice(0, 2)})}
                        className={`w-14 h-14 text-center text-2xl text-red-500 border-2 rounded-lg bg-red-50 focus:outline-none focus:border-red-400 ${errorFields.includes('regroupOne') ? 'flash-error' : 'border-red-200'}`}
                      />
                    </div>

                    {/* 2행: 빼어지는 수 */}
                    <div className="col-start-2 text-center w-full relative">
                      <span className={inputs.regroupTen !== '' ? 'crossed-out' : ''}>
                        {Math.floor(num1 / 10)}
                      </span>
                    </div>
                    <div className="col-start-3 text-center w-full">
                      {num1 % 10}
                    </div>

                    {/* 3행: 빼는 수 */}
                    <div className="col-start-1 text-center w-full text-slate-400">-</div>
                    <div className="col-start-2 text-center w-full">
                      {Math.floor(num2 / 10) || ''}
                    </div>
                    <div className="col-start-3 text-center w-full">
                      {num2 % 10}
                    </div>

                    {/* 가로줄 */}
                    <div className="col-span-3 w-full h-1 bg-slate-800 my-2"></div>

                    {/* 4행: 결과 입력 */}
                    <div className="col-start-2">
                      <input 
                        type="number" 
                        value={inputs.ansTen}
                        onChange={(e) => setInputs({...inputs, ansTen: e.target.value.slice(0, 1)})}
                        className={`w-16 h-16 text-center text-4xl text-blue-600 border-2 rounded-xl bg-blue-50 focus:outline-none focus:border-blue-400 ${errorFields.includes('ansTen') ? 'flash-error' : 'border-blue-200'}`}
                      />
                    </div>
                    <div className="col-start-3">
                      <input 
                        type="number" 
                        value={inputs.ansOne}
                        onChange={(e) => setInputs({...inputs, ansOne: e.target.value.slice(0, 1)})}
                        className={`w-16 h-16 text-center text-4xl text-blue-600 border-2 rounded-xl bg-blue-50 focus:outline-none focus:border-blue-400 ${errorFields.includes('ansOne') ? 'flash-error' : 'border-blue-200'}`}
                      />
                    </div>
                 </div>

                 {step === 3 && (
                   <button 
                     onClick={checkStep3}
                     className="mt-8 bg-blue-600 hover:bg-blue-700 text-white font-bold py-3 px-10 rounded-xl shadow-lg flex items-center text-xl transition-transform hover:scale-105 active:scale-95"
                   >
                     최종 정답 확인
                   </button>
                 )}
              </div>
            </div>
          </div>

          {/* 오답 팝업 */}
          {showErrorModal && (
            <div className="fixed inset-0 bg-black/40 flex items-center justify-center z-50 backdrop-blur-sm">
              <div className="bg-white p-8 rounded-3xl shadow-2xl text-center max-w-sm w-full mx-4 border-4 border-red-300">
                <XCircle className="w-20 h-20 text-red-500 mx-auto mb-4" />
                <h2 className="text-2xl font-bold text-slate-800 mb-2">아앗! 틀렸어요</h2>
                <p className="text-slate-600 mb-6 font-medium">어디가 틀렸는지 다시 한 번 꼼꼼히 확인해 볼까요?</p>
                <button 
                  onClick={closeErrorModal}
                  className="w-full bg-red-500 hover:bg-red-600 text-white font-bold py-3 rounded-xl transition-colors"
                >
                  다시 생각하기
                </button>
              </div>
            </div>
          )}

          {/* 성공 팝업 */}
          {showSuccessModal && (
            <div className="fixed inset-0 bg-black/40 flex items-center justify-center z-50 backdrop-blur-sm">
              <div className="bg-white p-8 rounded-3xl shadow-2xl text-center max-w-sm w-full mx-4 border-4 border-green-300">
                <div className="text-6xl mb-4">🎉</div>
                <h2 className="text-2xl font-bold text-slate-800 mb-2">정답입니다!</h2>
                <p className="text-slate-600 mb-6 font-medium">받아내림 뺄셈을 완벽하게 이해했군요!</p>
                <div className="space-y-3">
                  <button 
                    onClick={() => { setShowSuccessModal(false); generateProblem(mode); }}
                    className="w-full bg-blue-500 hover:bg-blue-600 text-white font-bold py-3 rounded-xl transition-colors flex justify-center items-center"
                  >
                    <ArrowRight className="w-5 h-5 mr-2" /> 다음 문제 풀기
                  </button>
                  <button 
                    onClick={() => { setShowSuccessModal(false); setScreen('home'); }}
                    className="w-full bg-slate-200 hover:bg-slate-300 text-slate-700 font-bold py-3 rounded-xl transition-colors flex justify-center items-center"
                  >
                    <RotateCcw className="w-5 h-5 mr-2" /> 홈으로 가기
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* O/X 퀴즈 팝업 (1단계 시작 시) */}
          {step === 1 && !hasAnsweredOx && (
            <div className="fixed inset-0 bg-black/60 flex items-center justify-center z-50 backdrop-blur-sm">
              <div className="bg-white p-8 rounded-3xl shadow-2xl text-center max-w-md w-full mx-4 border-4 border-blue-400">
                <div className="text-6xl mb-4">🤔</div>
                <h2 className="text-3xl font-bold text-slate-800 mb-4">잠깐! 생각해보자</h2>
                <div className="bg-blue-50 rounded-xl p-4 mb-6">
                  <p className="text-xl text-slate-700 font-medium leading-relaxed">
                    일의 자리 숫자 <br/>
                    <span className="text-3xl font-bold text-red-500 mx-1">{num1 % 10}</span>에서 
                    <span className="text-3xl font-bold text-blue-600 mx-1">{num2 % 10}</span>을(를) <br/>
                    바로 뺄 수 있을까요?
                  </p>
                </div>
                
                {oxFeedback && (
                  <div className="mb-4 text-red-500 font-bold animate-pulse text-lg">
                    {oxFeedback}
                  </div>
                )}

                <div className="flex gap-4 justify-center mt-4">
                  <button 
                    onClick={() => setOxFeedback(`아앗! ${num1 % 10}보다 ${num2 % 10}이(가) 더 커서 뺄 수 없어요.`)}
                    className="flex-1 bg-white border-4 border-green-400 hover:bg-green-50 text-green-600 font-bold py-6 rounded-2xl transition-transform hover:scale-105 active:scale-95 flex flex-col items-center justify-center"
                  >
                    <span className="text-4xl mb-1">O</span>
                    <span className="text-sm text-slate-600 mt-2">뺄 수 있어요</span>
                  </button>
                  <button 
                    onClick={() => {
                      setHasAnsweredOx(true);
                      setOxFeedback('');
                    }}
                    className="flex-1 bg-white border-4 border-red-400 hover:bg-red-50 text-red-600 font-bold py-6 rounded-2xl transition-transform hover:scale-105 active:scale-95 flex flex-col items-center justify-center"
                  >
                    <span className="text-4xl mb-1">X</span>
                    <span className="text-sm text-slate-600 mt-2">뺄 수 없어요</span>
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* 제작자 표시 (게임 화면) - 가운데 아래 */}
          <div className="fixed bottom-4 left-1/2 -translate-x-1/2 flex items-center bg-white/95 backdrop-blur px-6 py-3 rounded-full shadow-lg border-2 border-sky-200 z-50 transition-transform hover:scale-105">
            <img 
              src="경서 캐릭터(상반신).jpg" 
              alt="어썸오쌤" 
              className="w-12 h-12 rounded-full mr-3 object-cover border border-slate-200 shadow-sm"
              onError={(e) => { 
                e.target.onerror = null; 
                e.target.src = 'https://api.dicebear.com/7.x/notionists/svg?seed=Awesome'; 
              }}
            />
            <span className="font-extrabold text-slate-700 text-lg tracking-wide whitespace-nowrap">제작: 어썸오쌤</span>
          </div>

        </div>
      );
    }

    // ReactDOM.render 대신 React 18 방식 사용
    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
